# MNQSCALPV3 indicator source (Mai Language) — Jey paste 2026-10-20 08:23
Source of truth for V3. MNQSCALPV4 = this + typo fix `NOT(ISLASTBAR)` + V4 block (see V4_CURRENT_STATE section 5).

```
{====================================}
{ MNQSCALPV3 - CLEAN SIGNALS          }
{ MNQ 5 MINUTE CHART                  }
{ MOMENTUM + PULLBACK + EARLY WATCH   }
{ BACKTEST SIGNAL INDICATOR           }
{====================================}

{===== PARAMETERS =====}

RETEST_BARS:=8;
MOM_RANGE_MULT:=1.30;
MOM_VOL_MULT:=1.40;
RETEST_VOL_MULT:=1.10;

{===== CONFIRMED SWINGS =====}

PH:=REF(H,3)>REF(H,4) AND REF(H,3)>REF(H,5) AND REF(H,3)>REF(H,2) AND REF(H,3)>REF(H,1);

PL:=REF(L,3)<REF(L,4) AND REF(L,3)<REF(L,5) AND REF(L,3)<REF(L,2) AND REF(L,3)<REF(L,1);

{===== STRUCTURE LEVELS =====}

HIGH_LEVEL:=VALUEWHEN(PH,REF(H,3));
LOW_LEVEL:=VALUEWHEN(PL,REF(L,3));

{===== COMPLETED CANDLE BREAKS =====}

UP_RAW:=REF(HIGH_LEVEL,1)>0 AND REF(C,1)>REF(HIGH_LEVEL,1);

DN_RAW:=REF(LOW_LEVEL,1)>0 AND REF(C,1)<REF(LOW_LEVEL,1);

{===== FIRST BREAK PER SWING =====}

HAGE:=BARSLAST(REF(PH,1))+1;
LAGE:=BARSLAST(REF(PL,1))+1;

UP_FIRST:=UP_RAW AND COUNT(REF(C,1)>REF(HIGH_LEVEL,1),HAGE)=1;

DN_FIRST:=DN_RAW AND COUNT(REF(C,1)<REF(LOW_LEVEL,1),LAGE)=1;

BR_UP:=UP_FIRST;
BR_DN:=DN_FIRST;

{===== BREAKOUT LEVELS =====}

UP_LEVEL:=VALUEWHEN(BR_UP,REF(HIGH_LEVEL,1));
DN_LEVEL:=VALUEWHEN(BR_DN,REF(LOW_LEVEL,1));

{===== RETEST ZONES =====}

AVG_RANGE:=REF(MA(H-L,14),1);

UP_TOL:=VALUEWHEN(BR_UP,MAX(10,AVG_RANGE*0.75));
DN_TOL:=VALUEWHEN(BR_DN,MAX(10,AVG_RANGE*0.75));

UP_AGE:=BARSLAST(BR_UP);
DN_AGE:=BARSLAST(BR_DN);

LAST_DIR:=VALUEWHEN(BR_UP OR BR_DN,IF(BR_UP,1,-1));

UP_WINDOW:=LAST_DIR=1 AND UP_AGE>=1 AND UP_AGE<=RETEST_BARS;

DN_WINDOW:=LAST_DIR=-1 AND DN_AGE>=1 AND DN_AGE<=RETEST_BARS;

{===== CONFIRMATION CANDLES =====}

RANGE1:=REF(H-L,1);

BULL_CANDLE:=RANGE1>0 AND REF(C,1)>REF(O,1) AND REF(C-L,1)>=RANGE1*0.50;

BEAR_CANDLE:=RANGE1>0 AND REF(C,1)<REF(O,1) AND REF(H-C,1)>=RANGE1*0.50;

{===== BULLISH RETEST =====}

UP_TOUCH:=REF(L,1)<=UP_LEVEL+UP_TOL AND REF(H,1)>=UP_LEVEL-UP_TOL;

UP_RECLAIM:=REF(C,1)>UP_LEVEL;

UP_INVALID:=REF(C,1)<UP_LEVEL-UP_TOL;

UP_CONFIRM:=UP_WINDOW AND UP_TOUCH AND UP_RECLAIM AND BULL_CANDLE AND COUNT(UP_INVALID,MAX(UP_AGE,1))=0;

{===== BEARISH RETEST =====}

DN_TOUCH:=REF(H,1)>=DN_LEVEL-DN_TOL AND REF(L,1)<=DN_LEVEL+DN_TOL;

DN_REJECT:=REF(C,1)<DN_LEVEL;

DN_INVALID:=REF(C,1)>DN_LEVEL+DN_TOL;

DN_CONFIRM:=DN_WINDOW AND DN_TOUCH AND DN_REJECT AND BEAR_CANDLE AND COUNT(DN_INVALID,MAX(DN_AGE,1))=0;

{===== ONE RETEST PER BREAK =====}

UP_ENTRY_RAW:=UP_CONFIRM AND COUNT(UP_CONFIRM,MAX(UP_AGE,1))=1;

DN_ENTRY_RAW:=DN_CONFIRM AND COUNT(DN_CONFIRM,MAX(DN_AGE,1))=1;

{===== TRADING HOURS =====}

ENTRY_OK:=TIME>=1800 OR TIME<1630;

FLAT_NOW:=TIME>=1645 AND TIME<1700;

{===== COMPLETED CANDLE MOMENTUM =====}

MOM_RANGE:=REF(H-L,1);
MOM_AVG_RANGE:=REF(MA(H-L,14),2);

MOM_VOL:=REF(VOL,1);
MOM_AVG_VOL:=REF(MA(VOL,20),2);

RANGE_EXPANSION:=MOM_AVG_RANGE>0 AND MOM_RANGE>=MOM_AVG_RANGE*MOM_RANGE_MULT;

STRONG_VOLUME:=MOM_AVG_VOL>0 AND MOM_VOL>=MOM_AVG_VOL*MOM_VOL_MULT;

RETEST_VOLUME:=MOM_AVG_VOL>0 AND MOM_VOL>=MOM_AVG_VOL*RETEST_VOL_MULT;

STRONG_BULL:=MOM_RANGE>0 AND REF(C,1)>REF(O,1) AND REF(C-L,1)>=MOM_RANGE*0.75;

STRONG_BEAR:=MOM_RANGE>0 AND REF(C,1)<REF(O,1) AND REF(H-C,1)>=MOM_RANGE*0.75;

{===== MOMENTUM BREAKOUTS =====}

MOM_LONG:=BR_UP AND RANGE_EXPANSION AND STRONG_VOLUME AND STRONG_BULL;

MOM_SHORT:=BR_DN AND RANGE_EXPANSION AND STRONG_VOLUME AND STRONG_BEAR;

{===== FILTERED RETESTS =====}

RETEST_LONG:=UP_ENTRY_RAW AND RETEST_VOLUME;

RETEST_SHORT:=DN_ENTRY_RAW AND RETEST_VOLUME;

{===== EXECUTABLE ENTRY CANDIDATES =====}

RAW_LI:=(MOM_LONG OR RETEST_LONG) AND ENTRY_OK;

RAW_SI:=(MOM_SHORT OR RETEST_SHORT) AND ENTRY_OK;

{===== EARLY BULLISH WATCH =====}

FAST_MA:=REF(MA(C,5),1);

RECENT_HIGH:=REF(HHV(H,6),2);

EARLY_BULL:=REF(C,1)>RECENT_HIGH AND REF(C,1)>FAST_MA AND FAST_MA>REF(FAST_MA,1) AND REF(C,1)>REF(O,1) AND REF(C-L,1)>=REF(H-L,1)*0.65;

WATCH_SIG:=EARLY_BULL AND ENTRY_OK AND NOT(REF(EARLY_BULL,1));

{===== BULLISH WEAKNESS WATCH =====}

WEAK_LONG:=REF(C,1)<REF(LLV(L,3),2) AND REF(C,1)<FAST_MA;

WEAK_SIG:=WEAK_LONG AND NOT(REF(WEAK_LONG,1));

{===== VIRTUAL STATE FOR LEGACY EXITS =====}

LONG_MODE:=VALUEWHEN(RAW_LI OR DN_FIRST OR FLAT_NOW,IF(RAW_LI AND NOT(FLAT_NOW),1,0));

SHORT_MODE:=VALUEWHEN(RAW_SI OR UP_FIRST OR FLAT_NOW,IF(RAW_SI AND NOT(FLAT_NOW),1,0));

HAD_LONG:=REF(LONG_MODE,1)=1;
HAD_SHORT:=REF(SHORT_MODE,1)=1;

{===== ENTRY SIGNALS =====}

LI_SIG:=RAW_LI AND NOT(REF(RAW_LI,1));

SI_SIG:=RAW_SI AND NOT(REF(RAW_SI,1));

{===== LEGACY EXIT SIGNALS =====}

EL_SIG:=DN_FIRST AND HAD_LONG;
ES_SIG:=UP_FIRST AND HAD_SHORT;

{===== NUMERICAL ENGINE OUTPUTS =====}

LONG_ENTRY:IF(LI_SIG,1,0),NODRAW;

SHORT_ENTRY:IF(SI_SIG,1,0),NODRAW;

LONG_EXIT:IF(EL_SIG OR FLAT_NOW,1,0),NODRAW;

SHORT_EXIT:IF(ES_SIG OR FLAT_NOW,1,0),NODRAW;

BUY_SIGNAL:IF(LI_SIG,1,0),NODRAW;

SELL_SIGNAL:IF(EL_SIG OR FLAT_NOW,1,0),NODRAW;

{===== ADDITIONAL DIAGNOSTIC OUTPUTS =====}

MOM_LONG_SIGNAL:IF(MOM_LONG AND ENTRY_OK,1,0),NODRAW;

MOM_SHORT_SIGNAL:IF(MOM_SHORT AND ENTRY_OK,1,0),NODRAW;

EARLY_LONG_SIGNAL:IF(WATCH_SIG,1,0),NODRAW;

WEAK_LONG_SIGNAL:IF(WEAK_SIG,1,0),NODRAW;

ZERO_SIGNAL:0,NODRAW;




{===== FAST ALERTS + CONFIRMED TREND =====}

{ CURRENT CANDLE - PROVISIONAL ALERTS }

LIVE_HIGH:=REF(HHV(H,6),1);
LIVE_LOW:=REF(LLV(L,6),1);

LIVE_BULL:=ISLASTBAR AND ENTRY_OK AND C>LIVE_HIGH AND C>O AND C>REF(C,1);

LIVE_BEAR:=ISLASTBAR AND ENTRY_OK AND C<LIVE_LOW AND C<O AND C<REF(C,1);

{ COMPLETED CANDLE - EARLY CONFIRMATION }

FAST_RNG:=REF(H-L,1);
FAST_AVG:=REF(MA(H-L,14),2);

FAST_VOL:=REF(VOL,1);
FAST_VOL_AVG:=REF(MA(VOL,20),2);

FAST_BULL_BASE:=ENTRY_OK AND FAST_AVG>0 AND FAST_VOL_AVG>0 AND REF(C,1)>REF(HHV(H,6),2) AND REF(C,1)>REF(O,1) AND REF(C-L,1)>=FAST_RNG*0.65 AND FAST_RNG>=FAST_AVG*0.80 AND FAST_VOL>=FAST_VOL_AVG*0.80;

FAST_BEAR_BASE:=ENTRY_OK AND FAST_AVG>0 AND FAST_VOL_AVG>0 AND REF(C,1)<REF(LLV(L,6),2) AND REF(C,1)<REF(O,1) AND REF(H-C,1)>=FAST_RNG*0.65 AND FAST_RNG>=FAST_AVG*0.80 AND FAST_VOL>=FAST_VOL_AVG*0.80;

FAST_BULL_EVENT:=FAST_BULL_BASE AND NOT(REF(FAST_BULL_BASE,1));

FAST_BEAR_EVENT:=FAST_BEAR_BASE AND NOT(REF(FAST_BEAR_BASE,1));

{ STRONG MOMENTUM - COMPLETED CANDLE }

BULL_POWER:=ENTRY_OK AND MOM_AVG_RANGE>0 AND MOM_AVG_VOL>0 AND REF(C,1)>REF(HHV(H,6),2) AND STRONG_BULL AND MOM_RANGE>=MOM_AVG_RANGE*1.40 AND MOM_VOL>=MOM_AVG_VOL*1.25;

BEAR_POWER:=ENTRY_OK AND MOM_AVG_RANGE>0 AND MOM_AVG_VOL>0 AND REF(C,1)<REF(LLV(L,6),2) AND STRONG_BEAR AND MOM_RANGE>=MOM_AVG_RANGE*1.40 AND MOM_VOL>=MOM_AVG_VOL*1.25;

SB_EVENT:=BULL_POWER AND NOT(REF(BULL_POWER,1));

SL_EVENT:=BEAR_POWER AND NOT(REF(BEAR_POWER,1));

{ CONFIRMED BULLISH / BEARISH CANDIDATES }

BULL_CONF:=(LI_SIG OR FAST_BULL_EVENT) AND NOT(SB_EVENT);

BEAR_CONF:=(SI_SIG OR FAST_BEAR_EVENT) AND NOT(SL_EVENT);

{ EARLY CONFIRMED PRESSURE }

BULL_EARLY:=WATCH_SIG AND COUNT(WATCH_SIG,6)=1 AND NOT(BULL_CONF) AND NOT(SB_EVENT);

BEAR_EARLY:=WEAK_SIG AND FAST_BEAR_BASE AND NOT(BEAR_CONF) AND NOT(SL_EVENT);

{ CURRENT BAR LABELS - CAN CHANGE }

DRAWTEXT(LIVE_BULL,L-12,'BULLS'),COLOR93DCC3;

DRAWTEXT(LIVE_BEAR,H+12,'BEARS'),COLORF2D365;

{ HISTORICAL CONFIRMED LABELS }

DRAWTEXT(BULL_EARLY AND NOT(ISLASTBAR),L-16,'BULLS'),COLOR93DCC3;

DRAWTEXT(BEAR_EARLY AND NOT(ISLASTBAR),H+16,'BEARS'),COLORF2D365;

DRAWTEXT(BULL_CONF AND NOT(ISLASTBAR),L-24,'BULLISH'),COLOR25B77B;

DRAWTEXT(BEAR_CONF AND NOT(ISLASTBAR),H+24,'BEARISH'),COLORE79332;

DRAWTEXT(SB_EVENT AND N   OT(ISLASTBAR),L-32,'SB'),COLOR087A42;

DRAWTEXT(SL_EVENT AND NOT(ISLASTBAR),H+32,'SL'),COLORC93636;

{ DIAGNOSTIC OUTPUTS - NOT ENGINE ORDERS }

FAST_LONG_CANDIDATE:IF(FAST_BULL_EVENT,1,0),NODRAW;

FAST_SHORT_CANDIDATE:IF(FAST_BEAR_EVENT,1,0),NODRAW;```
